# .agh/loops/software-delivery.loop.yaml  (fork of built-in v4)
apiVersion: agh.loop/v1
kind: Loop
meta:
  name: software-delivery
  category: Engineering
concurrency: forbid              # forbid (default) | allow | queue · 2nd concurrent start → 409 + active run id
inputs:
  slug:           { type: string,  required: true }   # tasks are authored OUTSIDE the loop (C5)
  implementer:    { type: agent,   default: code_implementer }
  verify_command: { type: string,  default: "" }       # the project's OWN check (C2) · empty = disabled
  auto_commit:    { type: boolean, default: false }
  target_branch:  { type: string,  default: main }
contract:
  goal: "Ship every authored task under .compozy/tasks/{{ .inputs.slug }} · implemented, reviewed to acceptance, and verified by the project's own checks."
  definition_of_done: "All loaded tasks reached a terminal state, the acceptance judge approved with evidence, and the project verification passed."
  iteration_cap: 50            # generations (re-attempt rounds), not tasks · 0 = unbounded (watch)
  no_progress: { window: 3, hash_fields: [delivery_artifact, gate_verdict] }
  budget: { tokens: 0, wall_clock_sec: 0, on_exceeded: halt }   # 0 = unlimited · a SET budget is enforced
  terminal_states: [done, no_op, blocked, failed, exhausted, stalled]
graph:
  nodes:
    - { id: slug,       class: source,  kind: input, input_ref: slug }
    - { id: load_tasks, class: source,  kind: file-import,
        pattern: ".compozy/tasks/{{ .inputs.slug }}/task_*.md", parse: md_tasks }
    - { id: implement,  class: control, kind: fan-out,           # sequential, dependency order (blocks)
        collection: "{{ .nodes.load_tasks.output.tasks }}",
        batch_size: 1, max_parallel: 1, max_fan_out: 80 }   # 80 > ceiling 64
    - id: execute_task
      class: action
      kind: run-agent
      params:
        agent: "{{ .inputs.implementer }}"
        prompt: |
          Begin work on {{ .item.title }} using the installed cy-execute-task skill.
          …  (kickoff + required skills + repo AGENTS.md / PRD context folded)
          Task file: {{ .item.path }}
          {{ if .inputs.auto_commit }}Create one commit after verification.{{ else }}Leave changes uncommitted.{{ end }}
          ## Task Specification
          {{ .item.body }}
        output_schema:                                # schema fail → one FREE retry (ADR-021)
          type: object
          required: [status, summary]
          properties:
            status:  { enum: [completed, blocked] }
            summary: { type: string }
      session: { isolated: true }
      timeout: 45m
      retry: { max: 2 }
    - { id: collect, class: control, kind: collect }
    - id: review
      class: control
      kind: gate
      criteria:
        - { id: acceptance, type: agent-judge, agent: reviewer, rubric: "Diff satisfies every task's acceptance; cite evidence, emit stable blocking ids…" }
      verdict_policy: revise_until_clean   # requires a judge or human criterion
      on_result: { pass: verify, fail: next_generation }
      max_revisions: 3
    - id: verify
      class: control
      kind: gate
      criteria:
        - { id: project_check, type: command, check: "{{ .inputs.verify_command }}", expect: exit_zero }
      verdict_policy: fixed_passes         # the project's OWN check, never hardcoded (C2)
      on_result: { pass: approve, fail: next_generation }
    - id: approve
      class: control
      kind: gate
      criteria:
        - { id: ship, type: human, prompt: "Approve the delivered work for {{ .inputs.slug }} into {{ .inputs.target_branch }}?" }
      verdict_policy: fixed_passes
      on_result: { pass: done, fail: next_generation }
  edges:
    - { from: slug,             to: load_tasks }
    - { from: load_tasks,       to: implement }
    - { from: implement,        to: execute_task }
    - { from: execute_task,     to: collect }
    - { from: collect,          to: review }
    - { from: review,           to: verify }
    - { from: verify,           to: approve }
start: [ { kind: manual }, { kind: cli }, { kind: http }, { kind: uds }, { kind: native_tool }, { kind: schedule } ]
