name: release-train
graph:
  nodes:
    - { id: input, class: source, kind: input, input_ref: severity }
    - id: triage
      class: control
      kind: route
      routes:
        - { when: 'severity == "p0"', to: hotfix }
        - { when: 'severity == "p1"', to: standard }
      default: backlog
      on_eval_error: fail
    - id: standard
      class: action
      kind: run-agent
    - id: confirm-rollout
      class: control
      kind: ask
      params:
        prompt: "Which regions ship first?"
        expect: { type: object, required: [regions, canary] }
        responders: { agents: allow }
        expires: { after: 24h, route: backlog }
    - id: fan-services
      class: control
      kind: fan-out
      collection: "{{ .inputs.services }}"
      bind_as: service
      index_as: i
      strategy: { kind: best_effort, threshold: 66%, missing: acceptable }
    - id: apply-migration
      class: action
      kind: run-agent
      review:
        decisions: [approve, edit, reject, respond]
        on_reject: { route: backlog }
